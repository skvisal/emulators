import subprocess

# Run adb devices command
result = subprocess.run(['adb', 'devices'], capture_output=True, text=True)

# Split lines and strip any empty ones
lines = [line.strip() for line in result.stdout.strip().splitlines() if line.strip()]

# Get the last line (excluding the header line "List of devices attached")
if len(lines) > 1:
    last_device = lines[-1]
    print("Last device:", last_device)
else:
    print("No devices found.")
